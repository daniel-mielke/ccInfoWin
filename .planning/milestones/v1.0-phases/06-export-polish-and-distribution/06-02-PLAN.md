<!-- WinUI3Localizer initialization pattern from research -->
```csharp
// In App.xaml.cs OnLaunched, BEFORE creating MainWindow:
var stringsFolderPath = Path.Combine(AppContext.BaseDirectory, "Strings");
ILocalizer localizer = await new LocalizerBuilder()
    .AddStringResourcesFolderForLanguageDictionaries(stringsFolderPath)
    .SetOptions(options => { options.DefaultLanguage = "en-US"; })
    .Build();

// Runtime language switch (no restart):
await Localizer.Get().SetLanguage("de-DE");
```

<!-- .resw key naming for WinUI3Localizer x:Uid -->
- TextBlock text: `{Uid}.Text`
- Button content: `{Uid}.Content`
- AutomationProperties.Name: `{Uid}.[using:Microsoft.UI.Xaml.Automation]AutomationProperties.Name`
- ToggleSwitch.OnContent: `{Uid}.OnContent`
- ToggleSwitch.OffContent: `{Uid}.OffContent`
- ToolTipService.ToolTip: `{Uid}.[using:Microsoft.UI.Xaml.Controls]ToolTipService.ToolTip`

<!-- .csproj addition required -->
```xml
<ItemGroup>
  <Content Include="Strings\**\*.resw">
    <CopyToOutputDirectory>PreserveNewest</CopyToOutputDirectory>
  </Content>
</ItemGroup>
```

    var stringsFolderPath = Path.Combine(AppContext.BaseDirectory, "Strings");

    }
